TensorRT Engine Deployment Guide

Overview#

Building and using a TensorRT engine for inference involves two distinct phases: build time (optimization) and runtime (execution).

  1. Build time (optimization and compilation)
    1. Load model typically from format like ONNX.
    2. Parse model, understand network structure and weights.
    3. Optimize model, including layer, tensor fusion, precision calibration, kernel auto-tunning.
    4. Serialize model into a single, executable engine file.
  2. Runtime (Execution)
    1. Create Runtime instance.
    2. Deserialize engine into an ICudaEngine object
    3. Create context IExecutionContext from the engine, it manages a single inference session.
    4. Prepare buffer for input and output.
    5. Execute inference.
    6. Retrieve results.
    7. Free resources

Runtime Deployment Steps#

1. Create a Runtime Object#

Create an instance of the TensorRT runtime. This object is the entry point for all inference operations and is responsible for managing the execution of engines. It's a key part of the TensorRT API and is a lightweight object designed to be quickly instantiated.

nvinfer1::IRuntime* runtime = nvinfer1::createInferRuntime(gLogger);

Or

runtime = trt.Runtime(trt.Logger())

2. Deserialize the Engine#

The engine file (.engine or .plan) is a serialized binary file. The runtime's job is to read the file and deserialize it back into an ICudaEngine object. This process loads the pre-optimized network graph and weights into the GPU's memory.

std::ifstream file("my_model.engine", std::ios::binary);
file.seekg(0, file.end);
size_t size = file.tellg();
file.seekg(0, file.beg);
char* engine_data = new char[size];
file.read(engine_data, size);
nvinfer1::ICudaEngine* engine = runtime->deserializeCudaEngine(engine_data, size);

with open("my_model.engine", "rb") as f:
    engine = runtime.deserialize_cuda_engine(f.read())

3. Create an Execution Context#

The engine holds the model's core structure and weights, but to actually perform an inference run, need an execution context. This context is a lightweight object that holds the state for a single inference, including temporary GPU memory for intermediate tensors (activations). An engine can create multiple contexts, which is useful for running parallel inferences on the same model

nvinfer1::IExecutionContext* context = engine->createExecutionContext();

context = engine.create_execution_context()

4. Prepare Input and Output Buffers#

Before running the model, must allocate memory on the GPU for the input and output. Copy input data from CPU to GPU's memory.

void* buffers[2]; // Pointers for input and output
cudaMalloc(&buffers[0], input_size);
cudaMalloc(&buffers[1], output_size);
cudaMemcpy(buffers[0], host_input_data, input_size, cudaMemcpyHostToDevice);

d_input = cuda.mem_alloc(input_data.nbytes)
d_output = cuda.mem_alloc(output_data.nbytes)
cuda.memcpy_htod(d_input, input_data)

5. Run the Inference#

With the context and buffers ready, can execute the model. The enqueueV2 or enqueueV3 method of the execution context is used to launch the optimized kernels on the GPU. The is an asynchronous operation, so it will return immediately without waiting for the computation to finish.

context->executeV2(buffers);

context.execute_v2(bindings=[int(d_input), int(d_output)])

6. Retrieve the Results#

After the inference is complete, copy the results from GPU memory back to the CPU memory for post-processing and use. Use CUDA synchronization like cudaStreamSynchronize to ensure the GPU computation has finished before reading the data.

cudaMemcpy(host_output_data, buffers[1], output_size, cudaMemcpyDeviceToHost);

cuda.memcpy_dtoh(output_data, d_output)

7. Free Allocated Memory, Destroy Objects, Contexts#

Free the allocated memory for TensorRT engine, and destroy context and objects. This ensures GPU memory and resources are properly released, preventing memory leaks.

cudaFree(buffers[0]);
cudaFree(buffers[1]);

context->destroy();
engine->destroy();
runtime->destroy();

d_input.free()
d_output.free()

context.destroy()
engine.destroy()
runtime.destroy()

Example Code#

Below is an example loading a pre-built TensorRT engine and running inference.

#include <iostream>
#include <fstream>
#include <vector>
#include <string>
#include <numeric>
#include <cuda_runtime.h>

#include "NvInfer.h"
#include "NvOnnxParser.h"

// Custom TensorRT logger to catch warnings and errors
class Logger : public nvinfer1::ILogger {
public:
    void log(Severity severity, const char* msg) noexcept override {
        if (severity <= Severity::kINFO) { // Adjust severity level as needed
            std::cout << msg << std::endl;
        }
    }
};

// --- Helper Functions for CUDA Memory Management ---
void* safeCudaMalloc(size_t memSize) {
    void* deviceMem;
    cudaMalloc(&deviceMem, memSize);
    if (deviceMem == nullptr) {
        std::cerr << "CUDA memory allocation failed." << std::endl;
        exit(EXIT_FAILURE);
    }
    return deviceMem;
}

void checkCudaErrors(cudaError_t err) {
    if (err != cudaSuccess) {
        std::cerr << "CUDA error: " << cudaGetErrorString(err) << std::endl;
        exit(EXIT_FAILURE);
    }
}

// --- Main Function for Inference ---
int main() {
    Logger logger;

    // 1. Create a TensorRT Runtime object
    nvinfer1::IRuntime* runtime = nvinfer1::createInferRuntime(logger);
    if (!runtime) {
        std::cerr << "Failed to create IRuntime." << std::endl;
        return -1;
    }

    // 2. Load and deserialize the engine file
    std::string engine_path = "model.engine";
    std::ifstream file(engine_path, std::ios::binary);
    if (!file.good()) {
        std::cerr << "Error opening engine file: " << engine_path << std::endl;
        return -1;
    }

    file.seekg(0, std::ios::end);
    size_t size = file.tellg();
    file.seekg(0, std::ios::beg);
    std::vector<char> engine_data(size);
    file.read(engine_data.data(), size);

    nvinfer1::ICudaEngine* engine = runtime->deserializeCudaEngine(engine_data.data(), size);
    if (!engine) {
        std::cerr << "Failed to deserialize engine." << std::endl;
        return -1;
    }

    // 3. Create an Execution Context
    nvinfer1::IExecutionContext* context = engine->createExecutionContext();
    if (!context) {
        std::cerr << "Failed to create IExecutionContext." << std::endl;
        return -1;
    }

    // 4. Prepare host and device memory buffers
    // Assume a single-input, single-output network with known dimensions
    const int input_idx = engine->getBindingIndex("input_tensor");
    const int output_idx = engine->getBindingIndex("output_tensor");

    nvinfer1::Dims input_dims = engine->getBindingDimensions(input_idx);
    size_t input_size = std::accumulate(input_dims.d + 1, input_dims.d + input_dims.nbDims, engine->getBindingDataType(input_idx) == nvinfer1::DataType::kFLOAT ? sizeof(float) : 1, std::multiplies<size_t>()) * input_dims.d[0];

    nvinfer1::Dims output_dims = engine->getBindingDimensions(output_idx);
    size_t output_size = std::accumulate(output_dims.d + 1, output_dims.d + output_dims.nbDims, engine->getBindingDataType(output_idx) == nvinfer1::DataType::kFLOAT ? sizeof(float) : 1, std::multiplies<size_t>()) * output_dims.d[0];

    std::vector<float> host_input_data(input_size / sizeof(float));
    std::vector<float> host_output_data(output_size / sizeof(float));

    // Fill host_input_data with your actual input (e.g., pre-processed image)
    // For demonstration, we'll use a dummy input
    std::cout << "Preparing dummy input..." << std::endl;
    for (size_t i = 0; i < host_input_data.size(); ++i) {
        host_input_data[i] = static_cast<float>(rand()) / static_cast<float>(RAND_MAX);
    }

    // Allocate GPU memory
    void* device_buffers[2];
    device_buffers[input_idx] = safeCudaMalloc(input_size);
    device_buffers[output_idx] = safeCudaMalloc(output_size);

    // Copy input data from host to device
    checkCudaErrors(cudaMemcpy(device_buffers[input_idx], host_input_data.data(), input_size, cudaMemcpyHostToDevice));

    // 5. Run the inference
    std::cout << "Running inference..." << std::endl;
    bool status = context->executeV2(device_buffers);
    if (!status) {
        std::cerr << "Inference execution failed." << std::endl;
        return -1;
    }

    // 6. Retrieve the results
    checkCudaErrors(cudaMemcpy(host_output_data.data(), device_buffers[output_idx], output_size, cudaMemcpyDeviceToHost));

    // Print a few output values for verification
    std::cout << "Inference successful. First 5 output values:" << std::endl;
    for (int i = 0; i < 5; ++i) {
        std::cout << host_output_data[i] << " ";
    }
    std::cout << std::endl;

    // 7. Clean up and free resources
    checkCudaErrors(cudaFree(device_buffers[input_idx]));
    checkCudaErrors(cudaFree(device_buffers[output_idx]));

    context->destroy();
    engine->destroy();
    runtime->destroy();

    std::cout << "Cleaned up resources successfully." << std::endl;

    return 0;
}
